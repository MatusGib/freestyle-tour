# Developer panel Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A Dev chip in the game opens a panel where event winds, venues' usual winds and zones, the champion's wind and target, and an offset on every AI target can be tried as a draft, used from the next heat, then discarded or written to the repo's data files.

**Architecture:** `server/dev.mjs` reads the four data files, lays a draft (kept in `data/dev-draft.json`) over them and checks the result with each file's own checks, which name the field of any problem. Every server reader (the challenge, the champion's rooms, the venue catalog) asks it for the values in force, through getters on the server's context. `server/api/dev.mjs` serves the panel; `mods/dev/` draws it in the mods' lagoon look.

**Tech Stack:** Node 22 ES modules, `node:test`, no dependencies; browser ES modules for the mod.

Spec: `docs/superpowers/specs/2026-10-30-event-winds-design.md` ("The champion", "AI targets", "The draft", "The panel"). Build `2026-10-30-event-winds.md` first: this plan uses its `server/problem.mjs`, `server/event-winds.mjs` and `tour/event-winds.json`.

```js
import { existsSync, readFileSync } from "node:fs";
import { coastFor } from "../mods/venues/coast.js";
import { isGusts, isKnots } from "./event-winds.mjs";
import { problem } from "./problem.mjs";

const normalize = (value) =>
  String(value ?? "")
    .normalize("NFKD")
    .replace(/[̀-ͯ]/g, "")
    .toLowerCase()
    .replace(/[^a-z0-9]+/g, " ")
    .trim();

const between = (value, low, high) => typeof value === "number" && value >= low && value <= high;
const ZONE_RANGES = { gain: [0, 8], from: [0, 1000], over: [1, 1000], lee: [0, 6] };

// Checks a catalog and indexes it by id and by location spelling. Throws a
// problem() naming the first broken field.
export function indexVenueCatalog(catalog, name = "tour/venues.json") {
  if (!Number.isInteger(catalog?.version) || !Array.isArray(catalog.venues)) throw problem("venues", `${name} is not a valid venue catalog`);
  const byId = new Map();
  const byLocation = new Map();
  for (const venue of catalog.venues) {
    const path = `venues.${venue.id}`;
    if (!venue.id || !venue.name || !Array.isArray(venue.locations) || !venue.gameConditions) throw problem(path, `${name} has an incomplete venue`);
    const { knots, gustKnots } = venue.gameConditions;
    if (!isKnots(knots)) throw problem(`${path}.gameConditions.knots`, `${name} has unsupported conditions for ${venue.id}: knots must be a whole number from 12 to 28`);
    if (!isGusts(gustKnots)) throw problem(`${path}.gameConditions.gustKnots`, `${name} has unsupported conditions for ${venue.id}: gusts must be 0 or 2–6 kt`);
    for (const [field, [low, high]] of Object.entries(ZONE_RANGES)) {
      if (venue.windZones && !between(venue.windZones[field], low, high)) throw problem(`${path}.windZones.${field}`, `${name} has unsupported wind zones for ${venue.id}: ${field} must be from ${low} to ${high}`);
    }
    if (byId.has(venue.id)) throw problem(path, `${name} repeats venue id ${venue.id}`);
    byId.set(venue.id, venue);
    for (const location of venue.locations) {
      const key = normalize(location);
      const previous = byLocation.get(key);
      if (previous && previous.id !== venue.id) throw problem(path, `${name} maps ${location} to more than one venue`);
      byLocation.set(key, venue);
    }
  }
  return { version: catalog.version, venues: catalog.venues, byId, byLocation };
}

export function loadVenueCatalog(file) {
  if (!existsSync(file)) return null;
  return indexVenueCatalog(JSON.parse(readFileSync(file, "utf8")), file);
}
```

```js
// The Champion's challenge's wind: 21 kt by default, above the game's own 1v1
// rooms (15–19 kt), with each room's own gusts. The game gives a 1v1 room its
// wind from the room's id; a patch (patches.mjs) passes the champion's rooms
// through roomWind. His wind is data (server/champion.json, changeable in the
// developer panel): the server sails his heat in it, and the champion mod
// hands it to the game with setWind. Pure, so the server imports it too.

export const ROOM_PREFIX = "champion-";
export const WIND = { knots: 21, gustKnots: 6 };

let current = WIND;

// wind: { knots, gustKnots }, as GET /api/local/weekly answers.
export const setWind = (wind) => void (current = wind);

// conditions: the game's own wind for the room id.
export const roomWind = (id, conditions, wind = current) => (id.startsWith(ROOM_PREFIX) ? { ...conditions, knots: wind.knots } : conditions);
```

```js
const ROOM_GUSTS = [0, 4, 5, 6];

// Throws a problem() if his wind or target can't be sailed: 12–28 kt, gusts
// his rooms have, a target of 20–60 points.
export function checkChampion(champion, name = "server/champion.json") {
  if (!isKnots(champion?.wind?.knots)) throw problem("champion.wind.knots", `${name}: his wind must be a whole number of knots from 12 to 28`);
  if (!ROOM_GUSTS.includes(champion.wind.gustKnots)) throw problem("champion.wind.gustKnots", `${name}: his gusts must be 0, 4, 5 or 6 kt, as the game's 1v1 rooms have`);

  return champion;
}
```

```js
// The data the developer panel changes (event winds, venues, the champion and
// the AI's tuning) as their files have it, with the panel's draft laid over
// it. The draft is kept in data/ (git-ignored), so the repo's files change
// only when it is written to them. The server's readers all ask values().
import { existsSync, readFileSync, renameSync, writeFileSync } from "node:fs";
import { checkEventWinds, indexEventWinds } from "./event-winds.mjs";
import { problem } from "./problem.mjs";
import { NO_TUNING, checkTuning } from "./tuning.mjs";
import { indexVenueCatalog } from "./venues.mjs";
import { checkChampion } from "./weekly.mjs";

// Each part's file, from the repo's root, as the panel names them.
export const DATA_FILES = { eventWinds: "tour/event-winds.json", venues: "tour/venues.json", champion: "server/champion.json", tuning: "server/tuning.json" };

// The layout the data files are kept in, so a write's diff shows only the
// values that changed.
export const formatJson = (value) => `${JSON.stringify(value, null, 2)}\n`;

const read = (file) => (existsSync(file) ? JSON.parse(readFileSync(file, "utf8")) : null);
const NO_WINDS = { version: 1, events: [] };

// The files' data with a draft laid over it, and the parts of the draft for
// events or venues no longer in the files. draft: { eventWinds: { <event>:
// { knots, gustKnots, confidence, reported } }, venues: { <id>: {
// gameConditions, windZones } }, champion: { wind, target }, tuning: {
// aiTargetOffset } }, holding only changed values. An event without an entry
// gets one, inferred.
export function overlay(files, draft, tour) {
  const dropped = [];
  const eventWinds = structuredClone(files.eventWinds ?? NO_WINDS);
  for (const [id, change] of Object.entries(draft.eventWinds ?? {})) {
    const event = tour?.events.get(Number(id));
    if (!event) {
      dropped.push(`eventWinds.${id}`);
      continue;
    }
    let entry = eventWinds.events.find((e) => e.event === event.id);
    if (!entry) eventWinds.events.push((entry = { event: event.id, slug: event.slug, knots: null, gustKnots: 0, reported: "Set in the developer panel", confidence: "inferred", sources: [] }));
    Object.assign(entry, change);
  }
  const venues = structuredClone(files.venues);
  for (const [id, change] of Object.entries(draft.venues ?? {})) {
    const venue = venues?.venues.find((v) => v.id === id);
    if (!venue) {
      dropped.push(`venues.${id}`);
      continue;
    }
    if (change.gameConditions) Object.assign(venue.gameConditions, change.gameConditions);
    if (change.windZones) venue.windZones = { ...venue.windZones, ...change.windZones };
  }
  const champion = { ...structuredClone(files.champion), ...draft.champion, wind: { ...files.champion?.wind, ...draft.champion?.wind } };
  const tuning = { ...(files.tuning ?? NO_TUNING), ...draft.tuning };
  return { data: { eventWinds, venues, champion, tuning }, dropped };
}

// paths: each part's file ({ eventWinds, venues, champion, tuning }).
// draftStore: createStore(file, { draft: {} }), or null with the dev mod off,
// when the files' values hold. Broken files throw, so the server refuses to
// start. A saved draft that no longer fits is set aside (problem()) until the
// panel sends one that does.
export function createDevData({ paths, tour, draftStore = null, log = console.log }) {
  let files;
  let current;
  let setAside = null;
  const load = () => {
    files = Object.fromEntries(Object.entries(paths).map(([part, file]) => [part, read(file)]));
    if (!files.eventWinds) log(`No ${paths.eventWinds}: events sail in their venue's usual wind.`);
  };
  // Lays the draft over the files and checks the lot; throws a problem() at
  // the first broken field.
  const build = (draft) => {
    const { data, dropped } = overlay(files, draft, tour);
    if (tour) checkEventWinds(data.eventWinds, tour, paths.eventWinds);
    const values = {
      eventWinds: tour ? indexEventWinds(data.eventWinds) : null,
      venues: data.venues && indexVenueCatalog(data.venues, paths.venues),
      champion: checkChampion(data.champion, paths.champion),
      tuning: checkTuning(data.tuning, paths.tuning),
    };
    return { data, values, dropped };
  };
  const draft = () => draftStore?.data.draft ?? {};

  load();
  current = build({});
  try {
    current = build(draft());
  } catch (error) {
    setAside = error.message;
    log(`The developer panel's draft doesn't fit the files, so it is set aside: ${error.message}`);
  }
  if (current.dropped.length) log(`The developer panel's draft changes things no longer in the files: ${current.dropped.join(", ")}`);

  return {
    values: () => current.values,
    files: () => files,
    draft,
    dropped: () => current.dropped,
    problem: () => setAside,
    // Replaces the draft if it fits; otherwise throws a problem() and keeps
    // the last good one.
    setDraft(next) {
      if (!draftStore) throw problem("draft", "The dev mod is off.");
      const built = build(next);
      draftStore.update((d) => void (d.draft = next));
      current = built;
      setAside = null;
    },
    discard() {
      draftStore?.update((d) => void (d.draft = {}));
      current = build({});
      setAside = null;
    },
    // Writes the parts the draft changes, each through a temporary file, then
    // reloads them and clears the draft. Returns the parts written. If a write
    // fails, the error lists the parts already written, and the draft is kept.
    write() {
      const changes = draft();
      const { data } = build(changes);
      const written = [];
      for (const part of Object.keys(paths)) {
        if (!changes[part] || formatJson(data[part]) === formatJson(files[part])) continue;
        try {
          writeFileSync(`${paths[part]}.tmp`, formatJson(data[part]));
          renameSync(`${paths[part]}.tmp`, paths[part]);
        } catch (error) {
          load();
          current = build(changes);
          throw Object.assign(new Error(`Couldn't write ${paths[part]}: ${error.message}`), { written });
        }
        written.push(part);
      }
      load();
      draftStore?.update((d) => void (d.draft = {}));
      current = build({});
      return written;
    },
  };
}
```

```js
// Endpoints for the developer panel (the dev mod): the files' values, the
// draft and the events by venue; setting and discarding the draft; writing it
// into the repo's data files. ctx.data: createDevData().
import { DATA_FILES } from "../dev.mjs";
import { venueForLocation } from "../venues.mjs";
import { ok } from "./reply.mjs";

// The events grouped by venue, in the catalog's order, then the rest; each
// group in date order.
function eventsByVenue(tour, venues) {
  const groups = new Map((venues?.venues ?? []).map((v) => [v.id, { venue: { id: v.id, name: v.name }, events: [] }]));
  const other = { venue: null, events: [] };
  const events = [...(tour?.events.values() ?? [])].sort((a, b) => (a.start < b.start ? -1 : a.start > b.start ? 1 : a.id - b.id));
  for (const event of events) {
    const venue = venueForLocation(venues, event.location);
    (venue ? groups.get(venue.id) : other).events.push({ id: event.id, slug: event.slug, name: event.name, start: event.start });
  }
  return [...groups.values(), other].filter((group) => group.events.length);
}

export function devState({ data, tour }) {
  return ok({ files: data.files(), draft: data.draft(), dropped: data.dropped(), problem: data.problem(), events: eventsByVenue(tour, data.values().venues), dataFiles: DATA_FILES });
}

// A refused draft answers 400 with the reason and the field it is about.
export function setDraft(body, ctx) {
  try {
    ctx.data.setDraft(body ?? {});
  } catch (error) {
    if (!error.path) throw error;
    return { status: 400, body: { error: error.message, path: error.path } };
  }
  return devState(ctx);
}

export function discardDraft(ctx) {
  ctx.data.discard();
  return devState(ctx);
}

// Answers with the data files written, as the repo names them.
export function writeDraft(ctx) {
  try {
    const written = ctx.data.write().map((part) => DATA_FILES[part]);
    return ok({ ...devState(ctx).body, written });
  } catch (error) {
    return { status: 500, body: { error: error.message, written: (error.written ?? []).map((part) => DATA_FILES[part]) } };
  }
}
```

```js
// Pure helpers for the developer panel, shared with the tests. A draft holds
// only changed values, as POST /api/local/dev/draft takes it: { eventWinds:
// { <event>: {…} }, venues: { <id>: {…} }, champion: {…}, tuning: {…} }. A
// field is named by its path, as the server names a refused one:
// eventWinds.378.knots, venues.<id>.windZones.gain, champion.wind.knots,
// tuning.aiTargetOffset.

// The value at a dotted path, or undefined.
export const valueAt = (source, path) => path.split(".").reduce((node, key) => node?.[key], source);

// The files' value for a field (GET /api/local/dev's files), or undefined.
export function fileValue(files, path) {
  const [part, id, ...rest] = path.split(".");
  if (part === "eventWinds") return valueAt(files.eventWinds?.events.find((e) => String(e.event) === id), rest.join("."));
  if (part === "venues") return valueAt(files.venues?.venues.find((v) => v.id === id), rest.join("."));
  return valueAt(files[part], [id, ...rest].join("."));
}

// What a field shows: the draft's value, else the files'.
export function shownValue(draft, files, path) {
  const value = valueAt(draft, path);
  return value === undefined ? fileValue(files, path) : value;
}

// A copy of the draft with the field set; setting it back to the files'
// value takes the change out, with any group it leaves empty.
export function withChange(draft, files, path, value) {
  const next = structuredClone(draft);
  const keys = path.split(".");
  const chain = [next];
  for (const key of keys.slice(0, -1)) chain.push((chain.at(-1)[key] ??= {}));
  chain.at(-1)[keys.at(-1)] = value;
  if (value === fileValue(files, path)) {
    delete chain.at(-1)[keys.at(-1)];
    for (let i = keys.length - 2; i >= 0; i--) if (!Object.keys(chain[i + 1]).length) delete chain[i][keys[i]];
  }
  return next;
}

// How many values a draft changes.
export const changeCount = (draft) => (draft && typeof draft === "object" ? Object.values(draft).reduce((sum, child) => sum + changeCount(child), 0) : 1);
```

```js
// The developer panel: try event winds, venues' usual winds and zones, the
// champion's wind and target, and an offset on every AI target. Changes are a
// draft the local server keeps in data/ and lays over the files, used from
// the next heat; "Write to repo" writes them into the data files, "Discard"
// drops them.
import { createPopover, el } from "../lib/ui.js";
import { changeCount, fileValue, shownValue, valueAt, withChange } from "./dev.js";

// GET without a body, POST with one. A refused draft's error carries the
// field it is about (path).
async function request(url, body) {
  const init = body === undefined ? undefined : { method: "POST", headers: { "content-type": "application/json" }, body: JSON.stringify(body) };
  const res = await fetch(url, init);
  const answer = await res.json().catch(() => ({}));
  if (!res.ok) throw Object.assign(new Error(answer.error ?? `${url} answered ${res.status}`), { path: answer.path ?? null });
  return answer;
}

const gustOptions = (list) => list.map((g) => [g, g ? `${g} kt` : "Steady"]);
const GUSTS = gustOptions([0, 2, 3, 4, 5, 6]);
const ROOM_GUSTS = gustOptions([0, 4, 5, 6]); // the gusts the game's 1v1 rooms have
const CONFIDENCE = ["reported", "bounded", "inferred"].map((c) => [c, c]);
const TABS = ["Events", "Venues", "Champion", "AI"];
const changesText = (n) => (n ? `${n} change${n === 1 ? "" : "s"}` : "No changes");

export default {
  name: "dev",
  setup({ overlay, log }) {
    const holder = overlay("dev");
    holder.className = "ft-slot";
    const status = el("span", { className: "ft-chip-value", textContent: "No changes" });
    const chip = el("button", { type: "button", className: "ft-chip", ariaHasPopup: "dialog", onclick: () => toggle().catch((e) => log("could not open the developer panel", e)) },
      el("span", { className: "ft-chip-text" }, el("span", { className: "ft-chip-label", textContent: "Dev" }), status));
    holder.append(chip);
    const panel = createPopover("Developer panel", "dev");

    let state = null; // GET /api/local/dev's answer
    let draft = {};
    let tab = "Events";
    let eventId = null; // the event the Events tab shows
    let refused = null; // { path, error } for the last draft the server refused
    let note = "";
    let timer = null;
    const fields = new Map(); // path → the field's parts
    const count = el("span");
    const discard = el("button", { type: "button", className: "ft-button ft-button-small", textContent: "Discard", onclick: () => act(() => request("/api/local/dev/discard", {}), "Draft discarded.") });
    const write = el("button", { type: "button", className: "ft-button ft-button-small ft-button-primary", textContent: "Write to repo", onclick: () => act(() => request("/api/local/dev/write", {}), null) });
    const barReason = el("div", { className: "ft-dev-reason", role: "alert" });
    const noteLine = el("p", { className: "ft-muted" });

    // Marks changed fields with the file's value, shows why a value was
    // refused, and counts the changes.
    function refresh() {
      for (const [path, { row, file, reset, reason }] of fields) {
        const changed = valueAt(draft, path) !== undefined;
        row.classList.toggle("ft-dev-changed", changed);
        file.textContent = changed ? `file: ${fileValue(state.files, path) ?? "none"}` : "";
        reset.hidden = !changed;
        reason.textContent = refused?.path === path ? refused.error : "";
      }
      barReason.textContent = refused && !fields.has(refused.path) ? refused.error : "";
      const n = changeCount(draft);
      count.textContent = changesText(n);
      status.textContent = changesText(n);
      discard.disabled = !n;
      write.disabled = !n || !!refused;
      noteLine.textContent = note;
    }

    async function sendNow() {
      try {
        state = await request("/api/local/dev/draft", draft);
        refused = null;
      } catch (error) {
        refused = { path: error.path, error: error.message };
      }
    }
    // Sends the whole draft a moment after typing stops.
    function send() {
      clearTimeout(timer);
      timer = setTimeout(() => {
        timer = null;
        sendNow().catch((e) => log("could not send the draft", e)).finally(refresh);
      }, 400);
    }

    function change(path, value) {
      draft = withChange(draft, state.files, path, value);
      refresh();
      send();
    }

    // A field: a number, a choice or text, showing the draft's value or the
    // file's; when changed, the file's value beside it and a Reset button.
    function field(label, path, { kind = "number", options = null, step = 1 } = {}) {
      const value = shownValue(draft, state.files, path);
      let input;
      if (options) {
        input = el("select", {}, ...options.map(([v, text]) => el("option", { value: String(v), textContent: text })));
        input.value = String(value);
      } else input = el("input", { type: kind, value: value ?? "", step: String(step) });
      const read = () => {
        if (options) return typeof options[0][0] === "number" ? Number(input.value) : input.value;
        if (kind === "number") return input.value === "" ? null : Number(input.value);
        return input.value;
      };
      input.addEventListener(options ? "change" : "input", () => change(path, read()));
      const file = el("span", { className: "ft-muted" });
      const reset = el("button", { type: "button", className: "ft-button ft-button-small", textContent: "Reset", onclick: () => {
        change(path, fileValue(state.files, path));
        render();
      } });
      const reason = el("div", { className: "ft-dev-reason", role: "alert" });
      const row = el("label", { className: "ft-field" }, label, input, el("span", { className: "ft-dev-meta" }, file, reset), reason);
      fields.set(path, { row, file, reset, reason });
      return row;
    }

    function eventsTab() {
      eventId ??= state.events[0]?.events[0]?.id ?? null;
      if (eventId === null) return [el("p", { className: "ft-muted", textContent: "No tour data on this computer." })];
      const select = el("select", {}, ...state.events.map((group) => el("optgroup", { label: group.venue?.name ?? "Other venues" },
        ...group.events.map((e) => el("option", { value: String(e.id), textContent: e.name })))));
      select.value = String(eventId);
      select.onchange = () => {
        eventId = Number(select.value);
        render();
      };
      const at = (name) => `eventWinds.${eventId}.${name}`;
      return [
        el("label", { className: "ft-field" }, "Event", select),
        field("Knots (empty: the reports give none)", at("knots")),
        field("Gusts", at("gustKnots"), { options: GUSTS }),
        field("Confidence", at("confidence"), { options: CONFIDENCE }),
        field("What the reports say", at("reported"), { kind: "text" }),
      ];
    }
    function venuesTab() {
      return (state.files.venues?.venues ?? []).flatMap((venue) => {
        const at = (name) => `venues.${venue.id}.${name}`;
        return [
          el("h3", { className: "ft-division", textContent: venue.name }),
          field("Usual knots", at("gameConditions.knots")),
          field("Usual gusts", at("gameConditions.gustKnots"), { options: GUSTS }),
          ...(venue.windZones ? [
            field("Zone: knots more further out", at("windZones.gain")),
            field("Zone: starts this many metres out", at("windZones.from"), { step: 10 }),
            field("Zone: ramps in over metres", at("windZones.over"), { step: 10 }),
            field("Zone: knots less by the beach", at("windZones.lee")),
          ] : []),
        ];
      });
    }
    const championTab = () => [
      field("Knots", "champion.wind.knots"),
      field("Gusts", "champion.wind.gustKnots", { options: ROOM_GUSTS }),
      field("Target score", "champion.target", { step: 0.5 }),
      el("p", { className: "ft-muted", textContent: "His next room is sailed again with the new values." }),
    ];
    const aiTab = () => [field("Points added to every challenge rider's target", "tuning.aiTargetOffset", { step: 0.5 })];
    const TAB_CONTENT = { Events: eventsTab, Venues: venuesTab, Champion: championTab, AI: aiTab };

    function render() {
      fields.clear();
      const tabs = el("div", { className: "ft-dev-tabs", role: "tablist" }, ...TABS.map((name) => {
        const button = el("button", { type: "button", className: "ft-chip", role: "tab", textContent: name, onclick: () => {
          tab = name;
          render();
        } });
        button.setAttribute("aria-selected", String(tab === name));
        return button;
      }));
      panel.body.replaceChildren(
        el("p", { className: "ft-sub", textContent: "Changes are a draft the local server keeps, used from the next heat. Write to repo puts them in the data files." }),
        state.problem && el("p", { className: "ft-dev-reason", textContent: `Your saved draft doesn't fit the files: ${state.problem}` }),
        state.dropped.length > 0 && el("p", { className: "ft-muted", textContent: `Left out of the draft (no longer in the files): ${state.dropped.join(", ")}` }),
        tabs,
        el("div", { className: "ft-fields ft-dev-form" }, ...TAB_CONTENT[tab]()),
        el("div", { className: "ft-banner" }, count, el("span", { className: "ft-dev-meta" }, discard, write)),
        barReason,
        noteLine,
      );
      refresh();
    }

    // Discard or Write to repo, after sending a draft still waiting to go.
    async function act(call, done) {
      if (timer) {
        clearTimeout(timer);
        timer = null;
        await sendNow();
        if (refused) return refresh();
      }
      try {
        state = await call();
        draft = state.draft;
        refused = null;
        note = done ?? (state.written.length ? `Wrote ${state.written.join(", ")}. Commit them when you're happy.` : "Nothing to write.");
      } catch (error) {
        note = error.message;
      }
      render();
    }

    async function toggle() {
      if (panel.open) return panel.hide();
      panel.body.replaceChildren(el("p", { className: "ft-muted", textContent: "Loading…" }));
      panel.show(chip);
      try {
        state = await request("/api/local/dev");
      } catch (error) {
        panel.body.replaceChildren(el("p", { className: "ft-muted", textContent: `Couldn't load the developer panel: ${error.message}` }));
        throw error;
      }
      draft = state.draft;
      render();
    }

    // The chip counts a draft kept from an earlier session.
    request("/api/local/dev").then((answer) => {
      state = answer;
      draft = answer.draft;
      status.textContent = changesText(changeCount(draft));
    }).catch((error) => log("could not load the draft", error));
  },
};
```

# AI Riders Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** AI riders who sail real heats with the game's engine at the score their rating says, and a `challenge` mod to sail a solo heat against any rider of any event.

**Architecture:**
- A pinned copy of Freestajlo's planner sails heats in Node, recorded with the game's own recorder.
- A score model turns each rider's Glicko-2 rating into a seeded target.
- The local server sails a requested heat in a worker thread and caches it.
- The challenge mod plays the recording back through the opponent display during your next solo heat.
- A shared opponent slot keeps the ghost out of a challenge heat.

**Tech Stack:** Node 22+ (`node:worker_threads`, `node:test`), browser ES modules, the game's own engine and opponent display from the snapshot.

**Spec:** `docs/superpowers/specs/2026-10-24-ai-riders-design.md`

```js
// What a rider scores on the game's scale, from their Glicko-2 rating going
// into an event. Glicko's win chance is close to a normal curve with a scale
// of 295.6 rating points, so heat scores spread SPREAD points around an average
// that moves SPREAD·√2/295.6 points per rating point give the same win chances.

export const ANCHOR = { rating: 1965, score: 42 }; // the top 5% of men's ratings average 42
export const SPREAD = 6;
export const SLOPE = (SPREAD * Math.SQRT2) / 295.6;
export const FLOOR = 22; // the planner can't aim lower
export const CEILING = 50;

export const averageScore = (rating) => ANCHOR.score + (rating - ANCHOR.rating) * SLOPE;

// A 32-bit seed from any parts (FNV-1a), so a challenge always gets the same draw.
export function seedOf(...parts) {
  let hash = 0x811c9dc5;
  for (const char of parts.join("|")) {
    hash ^= char.codePointAt(0);
    hash = Math.imul(hash, 0x01000193) >>> 0;
  }
  return hash;
}

// Mulberry32: numbers in [0, 1) from a seed.
function generator(seed) {
  let a = seed >>> 0;
  return () => {
    a = (a + 0x6d2b79f5) >>> 0;
    let t = a;
    t = Math.imul(t ^ (t >>> 15), t | 1);
    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

// The score a rider aims for in one heat: their average plus a normal draw
// with spread SPREAD, clipped to FLOOR–CEILING.
export function heatTarget(rating, seed) {
  const next = generator(seed);
  const z = Math.sqrt(-2 * Math.log(1 - next())) * Math.cos(2 * Math.PI * next());
  return Math.min(CEILING, Math.max(FLOOR, averageScore(rating) + SPREAD * z));
}

// The standard normal distribution function (Abramowitz and Stegun 7.1.26).
function normal(x) {
  const t = 1 / (1 + (0.3275911 * Math.abs(x)) / Math.SQRT2);
  const erf = 1 - ((((1.061405429 * t - 1.453152027) * t + 1.421413741) * t - 0.284496736) * t + 0.254829592) * t * Math.exp(-(x * x) / 2);
  return x >= 0 ? (1 + erf) / 2 : (1 - erf) / 2;
}

// The model's chance that a rider rated ra outscores one rated rb.
export const winChance = (ra, rb) => normal((averageScore(ra) - averageScore(rb)) / (SPREAD * Math.SQRT2));
```

```js
// Sails a solo heat with the Freestajlo planner, recorded the way the game
// records one: the game's own Controls with its Recorder attached, one physics
// step at a time. The recording replays with mods/lib/replay.js.
import { createBrain } from "./freestajlo/brain.mjs";
import { wrap } from "./freestajlo/sim.mjs";

const LONGEST = 210 * 120; // a heat lasts 180 s; stop at 210 s whatever happens

// target: the heat score to aim for. seconds: stop early (for tests).
export function sailHeat(engine, conditions, target, { seconds = Infinity } = {}) {
  const controls = new engine.Controls();
  const recorder = new engine.Recorder();
  controls.recorder = recorder.record;
  const sim = wrap(engine, engine.heatStart(conditions, controls), controls);
  const brain = createBrain(engine, { target });
  const limit = Math.min(LONGEST, Math.round(seconds * 120));
  let steps = 0;
  while (!sim.state.heat.done && steps < limit) {
    sim.step(brain.update(sim.state, sim.act, controls));
    recorder.step();
    steps += 1;
  }
  const totals = engine.heatTotals(sim.state.heat);
  return {
    replay: recorder.replay(),
    score: totals.total,
    attempts: sim.state.heat.attempts,
    tricks: totals.counting.map((a) => ({ tack: a.tack, label: a.label, score: a.score })),
    steps,
    state: sim.state,
  };
}
```

```js
// A worker thread that sails AI heats for the local server. It loads the
// snapshot's engine once, sails each requested heat, and checks that the
// recording replays to the same score before answering.
import { parentPort, workerData } from "node:worker_threads";
import { createRunner } from "../../mods/lib/replay.js";
import { loadGame, openSnapshot } from "../snapshot.mjs";
import { sailHeat } from "./sail.mjs";

const engine = await loadGame(openSnapshot(workerData.root, workerData.snapshot), ["engine", "wind", "controls", "week"]);

parentPort.on("message", ({ id, conditions, target, seconds }) => {
  try {
    const { state, ...heat } = sailHeat(engine, conditions, target, { seconds });
    const runner = createRunner(engine, { conditions, replay: heat.replay });
    runner.advance(heat.steps);
    const replayed = engine.heatTotals(runner.state.heat).total;
    if (replayed !== heat.score) throw new Error(`the recording replays to ${replayed}, not ${heat.score}`);
    parentPort.postMessage({ id, heat });
  } catch (error) {
    parentPort.postMessage({ id, error: error.message });
  }
});
```

```js
// Sails AI heats in a worker thread, so the local server keeps answering while
// a heat is planned (about 20 seconds). The worker takes one heat at a time, in
// the order asked, and starts on the first request.
import { Worker } from "node:worker_threads";

export function createSailer({ root, snapshot }) {
  let worker = null;
  let next = 0;
  const waiting = new Map();
  const failAll = (error) => {
    for (const { reject } of waiting.values()) reject(error);
    waiting.clear();
    worker = null;
  };
  const start = () => {
    worker = new Worker(new URL("./worker.mjs", import.meta.url), { workerData: { root, snapshot } });
    worker.unref();
    worker.on("message", ({ id, heat, error }) => {
      const job = waiting.get(id);
      waiting.delete(id);
      if (error) job.reject(new Error(error));
      else job.resolve(heat);
    });
    worker.on("error", failAll);
  };
  return {
    // Resolves to { replay, score, attempts, tricks, steps }.
    sail(conditions, target, { seconds } = {}) {
      if (!worker) start();
      const id = next++;
      return new Promise((resolve, reject) => {
        waiting.set(id, { resolve, reject });
        worker.postMessage({ id, conditions, target, seconds });
      });
    },
    close() {
      return worker?.terminate();
    },
  };
}
```

```js
// Sails AI heats at several targets in several weeks' solo winds and reports
// how close the scores land. The bar: within 1.0 on average, never more than 3.
//   node tools/ai-check.mjs [--weeks 5] [--targets 22,28,34,40,46]
import { availableParallelism } from "node:os";
import { fileURLToPath, pathToFileURL } from "node:url";
import { parseArgs } from "node:util";
import { createSailer } from "../lib/ai/sailer.mjs";
import { loadGame, openSnapshot } from "../lib/snapshot.mjs";

export const BAR = { mean: 1.0, worst: 3 };
const WEEK = 7 * 24 * 3600 * 1000;

// results: [{ target, score }]. Returns the mean and worst miss and whether
// they meet the bar.
export function judge(results) {
  const misses = results.map((r) => Math.abs(r.score - r.target));
  const mean = misses.reduce((sum, m) => sum + m, 0) / misses.length;
  const worst = Math.max(...misses);
  return { mean, worst, pass: mean <= BAR.mean && worst <= BAR.worst };
}

async function main() {
  const { values } = parseArgs({ options: { weeks: { type: "string", default: "5" }, targets: { type: "string", default: "22,28,34,40,46" } } });
  const root = fileURLToPath(new URL("..", import.meta.url));
  const snapshot = openSnapshot(root);
  const game = await loadGame(snapshot, ["week"]);
  const targets = values.targets.split(",").map(Number);
  const jobs = [];
  for (let back = 0; back < Number(values.weeks); back++) {
    const weekStart = game.weekStart(Date.now() - back * WEEK);
    for (const target of targets) jobs.push({ weekStart, conditions: game.weekConditions(weekStart), target });
  }
  const lanes = Math.max(1, Math.min(jobs.length, availableParallelism() - 1));
  console.log(`Sailing ${jobs.length} heats in ${lanes} worker threads…`);
  const results = [];
  const queue = [...jobs];
  await Promise.all(Array.from({ length: lanes }, async () => {
    const sailer = createSailer({ root, snapshot: snapshot.name });
    while (queue.length) {
      const job = queue.shift();
      const heat = await sailer.sail(job.conditions, job.target);
      results.push({ ...job, score: heat.score });
      const c = job.conditions;
      console.log(`  ${new Date(job.weekStart).toISOString().slice(0, 10)} (${c.knots} kt +${c.gustKnots})  target ${job.target}  scored ${heat.score.toFixed(2)}`);
    }
    await sailer.close();
  }));
  const { mean, worst, pass } = judge(results);
  console.log(`\nMean miss ${mean.toFixed(2)}, worst ${worst.toFixed(2)} (bar: ${BAR.mean} and ${BAR.worst}): ${pass ? "pass" : "FAIL"}`);
  if (!pass) process.exitCode = 1;
}

if (import.meta.url === pathToFileURL(process.argv[1]).href) {
  main().catch((error) => {
    console.error(error.message);
    process.exit(1);
  });
```

```js
// The game draws one opponent. Mods that want it take turns: the one holding
// the slot shows its rider, and the others leave the opponent display alone.
let owner = null;

// True when who now holds the slot (or already did).
export function claimOpponent(who) {
  if (owner && owner !== who) return false;
  owner = who;
  return true;
}

export function releaseOpponent(who) {
  if (owner === who) owner = null;
}
```

```js
// Pure helpers for the challenge mod, shared with the tests. A challenge is an
// AI heat as POST /api/local/ai-heats returns it.

// A solo heat at its first step: practice has no heat, and 1v1 heats run on a
// competition clock.
export const isSoloHeatStart = (state, game) => state.heat.enabled && state.totalTime === 0 && !game.competitionClock;

// A sample for the game's opponent display, with the pose nested the way its
// 1v1 transport sends it.
export const aiSample = (pose, now) => ({ pose, updatedAt: now, streamId: "challenge" });

export const riderLine = (r) => `${r.place}. ${r.pseudonym} · ${r.average.toFixed(1)}`;
export const planningText = (pseudonym) => `Planning ${pseudonym}'s heat…`;
export const nextHeatText = (heat) => `Next solo heat: vs ${heat.pseudonym} (target ${heat.target.toFixed(1)})`;

// A challenge sailed in an earlier week's wind has to be planned again.
export const isStale = (heat, weekStart) => heat.weekStart !== weekStart;

export function resultText(yours, heat) {
  const winner = yours > heat.score ? "you win" : yours < heat.score ? `${heat.pseudonym} wins` : "a draw";
  return `You ${yours.toFixed(2)} · ${heat.pseudonym} ${heat.score.toFixed(2)}: ${winner}`;
}
```

```js
// Sail a solo heat against an AI rider. Pick an event and a rider; the local
// server sails that rider's heat in this week's solo wind (about 20 seconds),
// and your next solo heat shows them as the opponent. At the end, both totals
// and the winner. The AI's heat never affects your physics or score.
import { claimOpponent, releaseOpponent } from "../lib/opponent.js";
import { createRunner } from "../lib/replay.js";
import { aiSample, isSoloHeatStart, isStale, nextHeatText, planningText, resultText, riderLine } from "./challenge.js";

const PANEL_STYLE =
  "position:fixed;left:12px;bottom:12px;z-index:100000;width:360px;max-height:75vh;overflow:auto;" +
  "font:12px/1.5 system-ui,sans-serif;background:rgba(10,40,40,.94);color:#e8f4f2;padding:10px 12px;border-radius:10px";

const getJson = async (url, init) => {
  const res = await fetch(url, init);
  const body = await res.json().catch(() => ({}));
  if (!res.ok) throw new Error(body.error ?? `${url} answered ${res.status}`);
  return body;
};
const postJson = (url, body) => getJson(url, { method: "POST", headers: { "content-type": "application/json" }, body: JSON.stringify(body) });
const el = (tag, props = {}, ...children) => {
  const node = Object.assign(document.createElement(tag), props);
  node.append(...children.filter((child) => child !== null));
  return node;
};

export default {
  name: "challenge",
  setup({ engine, overlay, whenGame, onStep, log }) {
    const holder = overlay("challenge");
    holder.style.cssText = "font:12px/1.4 system-ui,sans-serif;background:rgba(10,40,40,.85);color:#e8f4f2;padding:4px 10px;border-radius:8px;pointer-events:auto";
    const status = el("span");
    holder.append(el("button", { type: "button", textContent: "Challenge", onclick: () => openPanel().catch((e) => log("could not open the challenge panel", e)) }), status);
    const panel = el("div", { hidden: true });
    panel.style.cssText = PANEL_STYLE;
    document.body.append(panel);

    let game = null;
    let armed = null; // the planned AI heat for the next solo heat
    let planning = null; // { rider, pseudonym, event } while the server sails
    let session = null; // the game state of the current session
    let ai = null; // the runner replaying the AI heat in this session
    const say = (text) => void (status.textContent = text ? ` ${text}` : "");

    function arm(heat) {
      armed = heat;
      claimOpponent("challenge");
      say(nextHeatText(heat));
    }
    function disarm() {
      armed = null;
      releaseOpponent("challenge");
    }
    async function plan(rider, pseudonym, event) {
      planning = { rider, pseudonym, event };
      say(planningText(pseudonym));
      try {
        arm(await postJson("/api/local/ai-heats", { rider, event }));
      } catch (error) {
        log("could not plan the heat", error);
        say("Couldn't plan the heat");
        status.append(" ", el("button", { type: "button", textContent: "Retry", onclick: () => plan(rider, pseudonym, event) }));
      } finally {
        planning = null;
      }
    }

    async function openPanel() {
      const seasons = await getJson("/api/local/tour/seasons");
      const seasonSelect = el("select", {}, ...seasons.map((s) => el("option", { value: s.season, textContent: s.season })));
      const eventSelect = el("select");
      const riders = el("div");
      const showEvent = async () => {
        const event = await getJson(`/api/local/tour/events/${eventSelect.value}`);
        riders.replaceChildren(
          ...Object.entries(event.divisions).flatMap(([division, list]) => [
            el("strong", { textContent: division === "men" ? "Men" : "Women" }),
            el("ol", { style: "list-style:none;padding:0;margin:2px 0 8px" }, ...list.map((r) =>
              el("li", {}, `${riderLine(r)} `, el("button", { type: "button", textContent: "Challenge", onclick: () => {
                panel.hidden = true;
                plan(r.rider, r.pseudonym, event.id);
              } })))),
          ]),
        );
      };
      const showSeason = () => {
        const season = seasons.find((s) => String(s.season) === seasonSelect.value);
        eventSelect.replaceChildren(...season.events.map((e) => el("option", { value: e.id, textContent: e.name })));
        return showEvent();
      };
      seasonSelect.onchange = () => showSeason().catch((e) => log("could not list the event", e));
      eventSelect.onchange = () => showEvent().catch((e) => log("could not list the event", e));
      const cancel = armed && el("p", {}, `${nextHeatText(armed)} `, el("button", { type: "button", textContent: "Cancel", onclick: () => { disarm(); say(""); panel.hidden = true; } }));
      panel.replaceChildren(
        ...[
          el("strong", { textContent: "Challenge " }),
          el("button", { type: "button", textContent: "Close", onclick: () => void (panel.hidden = true) }),
          cancel,
          el("p", {}, seasonSelect, " ", eventSelect),
          riders,
        ].filter(Boolean),
      );
      panel.hidden = false;
      await showSeason();
    }

    whenGame().then((found) => {
      game = found;
      onStep((state) => {
        if (state !== session) {
          session = state;
          if (ai) game.clearOpponent();
          ai = null;
          if (isSoloHeatStart(state, game)) {
            if (armed && isStale(armed, engine.weekStart(Date.now()))) {
              const { rider, pseudonym, event } = armed;
              disarm();
              plan(rider, pseudonym, event);
            } else if (armed) {
              ai = createRunner(engine, { conditions: armed.conditions, replay: armed.replay });
              game.showOpponent(armed.pseudonym, () => Date.now());
              say(`Racing ${armed.pseudonym}`);
              log(`racing ${armed.pseudonym}, target ${armed.target}`);
            } else if (planning) say(`${planningText(planning.pseudonym)} This heat has no opponent.`);
          }
        }
        if (!ai) return;
        if (!ai.done) {
          ai.advance(1);
          game.receiveOpponent(aiSample(engine.riderPose(ai.state), performance.now()));
        }
        if (state.heat.done) {
          say(resultText(engine.heatTotals(state.heat).total, armed));
          game.clearOpponent();
          ai = null;
          disarm();
        }
      });
    });
  },
};
```
